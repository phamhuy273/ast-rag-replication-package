\noindent{\large\textbf{1. Bảng 4 (Table IV): Kết quả Kiểm định Giả thuyết Thống kê (Wilcoxon Signed-Rank Test)}}\\[0.05cm]
\input{tables/table_T4.tex}
\vspace{-0.1cm}
{\small
\noindent\textbf{Tóm tắt Kết luận Thực nghiệm \& Ý nghĩa Khoa học (Bảng 4):}
\begin{itemize}[nosep,topsep=2pt,partopsep=0pt,itemsep=2pt]
    \item \textbf{Kiểm định Giả thuyết Cốt lõi (Comp 1):} So sánh phương pháp đề xuất toàn diện (AST+Header) với baseline dòng toàn diện (Line No-Header) cho thấy $\Delta = -0.0930$ (95\% CI $[-0.164, -0.023]$, $p_{\mathrm{raw}} = 0.0219$, $p_{\mathrm{Holm}} = 0.0438$, $W = 78.0$). Có ý nghĩa thống kê, phản ánh Line No-Header đạt hiệu năng cao hơn AST+Header.
    \item \textbf{Hiệu ứng Cú pháp Thuần túy (Comp 2):} Khi loại bỏ Context Header ở cả 2 bên, phân mảnh theo AST thuần túy kém hơn Line cố định có ý nghĩa thống kê rất cao ($\Delta = -0.2390$, $p_{\mathrm{Holm}} < 0.001$, Line thắng 23/25 JD). Điều này chứng minh cấu trúc cây cú pháp đơn thuần không tự động giúp mô hình embedding hiểu code tốt hơn.
    \item \textbf{Context Header là Yếu tố Cứu vãn (Comp 3):} Việc gắn thêm Context Header (tên class, gói, chữ ký) giúp AST tăng vọt $\Delta = +0.1459$ ($p_{\mathrm{Holm}} = 0.0001$, AST có Header thắng 23/25 JD). Header chính là nguồn cung cấp thông tin ngữ nghĩa cốt lõi.
    \item \textbf{Đối chứng Mở rộng (Comp 5):} Khi cả 2 phương pháp đều được trang bị Context Header, Line+Header dẫn trước AST+Header có ý nghĩa thống kê ($\Delta = +0.1339$, $p_{\mathrm{raw}} = 0.0003$, $W = 36.0$).
\end{itemize}
}
